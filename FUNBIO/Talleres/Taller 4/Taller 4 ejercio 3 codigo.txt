#include <Wire.h>
const int MPU_ADDR = 0x68;
const int BUZZER_PIN = 8;
const float UMBRAL_CAIDA_LIBRE = 0.4;
const float UMBRAL_IMPACTO     = 2.0;
int16_t AcX, AcY, AcZ;
void setup() {
  Serial.begin(9600);
  Wire.begin();
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x6B);
  Wire.write(0);
  Wire.endTransmission(true);
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x1C);
  Wire.write(0x00);
  Wire.endTransmission(true);
  pinMode(BUZZER_PIN, OUTPUT);
  noTone(BUZZER_PIN);
  Serial.println("MPU6050 Inicializado. Sistema listo para monitorear caidas.");
}
void loop() {
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x3B); 
  Wire.endTransmission(false);
  Wire.requestFrom(MPU_ADDR, 6, true);
  AcX = Wire.read() << 8 | Wire.read();
  AcY = Wire.read() << 8 | Wire.read();
  AcZ = Wire.read() << 8 | Wire.read();
  float ax = AcX / 16384.0;
  float ay = AcY / 16384.0;
  float az = AcZ / 16384.0;
  float magAceleracion = sqrt(ax * ax + ay * ay + az * az);
  Serial.print("Acel Total: ");
  Serial.print(magAceleracion);
  Serial.println(" g");
  if (magAceleracion < UMBRAL_CAIDA_LIBRE || magAceleracion >= UMBRAL_IMPACTO) {
    Serial.println("--- ¡ALERTA DE CAÍDA DETECTADA! ---");
    for (int i = 0; i < 5; i++) {
      tone(BUZZER_PIN, 1000); 
      delay(100);
      tone(BUZZER_PIN, 1500); 
      delay(100);
    }
    noTone(BUZZER_PIN);
  } else {
    noTone(BUZZER_PIN);
  }
  delay(100);
}
